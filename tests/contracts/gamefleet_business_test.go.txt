package platform

import (
	"context"
	"encoding/json"
	"net/http"
	"net/http/httptest"
	"testing"

	nakamabridge "github.com/xuhuanhello/nakama-agones/pkg/gamefleet"
)

// This overlay test uses GameFleet's real private handler, isolated SQLite
// ledger, generated fixture keys and signed room receipts, not JSON stubs.
func TestNakamaGameFleetClientContract(t *testing.T) {
	x := newTicketHTTPFixture(t, true)
	s := httptest.NewServer(x.b.handler)
	defer s.Close()
	var app, revision, region string
	if err := x.x.f.s.DB.QueryRow(`SELECT d.application_id,p.revision_id,c.region FROM capacity_placements p JOIN catalog_deployments d ON d.id=p.deployment_id JOIN clusters c ON c.id=p.cluster_id WHERE p.id=?`, x.x.placement).Scan(&app, &revision, &region); err != nil {
		t.Fatal(err)
	}
	c, err := nakamabridge.NewClient(nakamabridge.Config{URL: s.URL, Key: x.b.key, ApplicationID: app, PlacementID: x.x.placement, RevisionID: revision, Region: region, Compatibility: "game-v1"})
	if err != nil {
		t.Fatal(err)
	}
	defer c.Close()
	ctx := context.Background()
	if err = c.CheckScope(ctx); err != nil {
		t.Fatal("scope", err)
	}
	current, err := c.Current(ctx, "player-one")
	if err != nil || current.Current == nil || current.Current.Reservation.AllocationID != x.command.AllocationID || current.Current.ConnectionGeneration != 0 {
		t.Fatal("current", err)
	}
	replay, err := c.Reserve(ctx, "ticket-http-reservation", []string{"player-one", "player-two"})
	if err != nil || !replay.Replay || replay.Reservation.AllocationID != x.command.AllocationID {
		t.Fatal("reservation replay", err)
	}
	_, err = c.Current(ctx, "not-allowlisted")
	if err == nil {
		t.Fatal("unlisted participant accepted")
	}
	join, err := c.Issue(ctx, x.command.AllocationID, "player-one", "contract-join-attempt", 0, false)
	if err != nil || join.Replay || join.Assignment.Endpoint == nil || join.Assignment.Ticket.State != "issued" {
		t.Fatal("join", err)
	}
	again, err := c.Issue(ctx, x.command.AllocationID, "player-one", "contract-join-attempt", 0, false)
	if err != nil || !again.Replay || again.Assignment.Ticket != join.Assignment.Ticket {
		t.Fatal("ticket replay", err)
	}
	// The existing helper consumes a real signed seat ticket; it increments
	// the same player's persisted generation without changing the reservation.
	consumeCurrentTicketForTest(t, x)
	current, err = c.Current(ctx, "player-one")
	if err != nil || current.Current.ConnectionGeneration != 1 {
		t.Fatal("consumed generation", err)
	}
	superseded, err := c.Issue(ctx, x.command.AllocationID, "player-one", "contract-join-attempt", 0, false)
	if err != nil || !superseded.Replay || superseded.Assignment.Ticket.State != "superseded" || superseded.Assignment.Endpoint != nil {
		t.Fatal("superseded replay", err)
	}
	resume, err := c.Issue(ctx, x.command.AllocationID, "player-one", "contract-resume-attempt", 1, true)
	if err != nil || resume.Assignment.Ticket.Generation != 2 {
		t.Fatal("resume", err)
	}
	// Endpoint evidence can become stale independently of the held reservation.
	stale := x.observation
	stale.Complete = false
	x.saveObservation(t, stale)
	if _, err = c.Issue(ctx, x.command.AllocationID, "player-one", "contract-resume-attempt", 1, true); err == nil {
		t.Fatal("stale endpoint bypassed")
	}
	current, err = c.Current(ctx, "player-one")
	if err != nil || current.Current == nil {
		t.Fatal("stale endpoint lost held room", err)
	}
	x.saveObservation(t, x.observation)
	if _, err = c.Issue(ctx, x.command.AllocationID, "player-one", "contract-resume-attempt", 1, true); err != nil {
		t.Fatal("same attempt not recoverable", err)
	}
	cancelled, err := c.Cancel(ctx, x.command.AllocationID)
	if err != nil || !cancelled.Reservation.CancellationRequested {
		t.Fatal("cancel intent", err)
	}
	current, err = c.Current(ctx, "player-two")
	if err != nil || current.Current == nil || !current.Current.Reservation.CancellationRequested {
		t.Fatal("cancel hid room", err)
	}
	// Revocation remains authoritative after the client passed startup preflight.
	x.x.f.call(http.MethodDelete, "/api/v1/business-callers/"+x.b.caller+"/keys/"+x.b.keyID, nil, http.StatusOK)
	if _, err = c.Current(ctx, "player-one"); err == nil {
		t.Fatal("revoked caller key remained usable")
	}
	var held int
	if err = x.x.f.s.DB.QueryRow("SELECT count(*) FROM capacity_room_holds WHERE state='held'").Scan(&held); err != nil || held != 1 {
		t.Fatal("lookup/retry/cancel changed held room count", err)
	}
	// Keep the import's exact client JSON forms exercised as normal encoding.
	if _, err = json.Marshal(current); err != nil {
		t.Fatal(err)
	}
}
